# ── API Backend ───────────────────────────────────────────────
server {
    listen 80;
    server_name api.vetnetcodip.com;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name api.vetnetcodip.com;

    ssl_certificate     /etc/letsencrypt/live/vetnetcodip.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/vetnetcodip.com/privkey.pem;

    location / {
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_cache_bypass $http_upgrade;
    }
}

    return 301 https://$host$request_uri;
}

    listen 443 ssl;
    server_name pagos.vetnetcodip.com;

    ssl_certificate_key /etc/letsencrypt/live/vetnetcodip.com/privkey.pem;
    # API del pagos-saas
    location /api/ {
        proxy_pass http://localhost:3030/api/;
        proxy_http_version 1.1;

        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
    # Archivos estáticos
    location ~* \.(js|css|html)$ {

        add_header Expires "0";

        try_files $uri $uri/ =404;
    }
    # Todo lo demás — SPA
    location / {
        root /var/www/app_veterinaria/pagos-saas/public;
        index index.html;
        try_files $uri $uri.html $uri/ /index.html;
    }
}

# ── Places Scraper / Embudo Ventas CRM ───────────────────────
server {
    listen 80;
    server_name places-scraper.vetnetcodip.com;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name places-scraper.vetnetcodip.com;

    ssl_certificate     /etc/letsencrypt/live/vetnetcodip.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/vetnetcodip.com/privkey.pem;

    root /var/www/places-scraper/public;
    index index.html;

    location ^~ /socket.io/ {
        proxy_pass http://localhost:3010;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_cache_bypass $http_upgrade;
        proxy_buffering off;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
        proxy_connect_timeout 60s;
    }

    location ^~ /api/ {
        proxy_pass http://localhost:3010;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 120s;
    }

    location ^~ /uploads/ {
        alias /var/www/places-scraper/uploads/;
        expires 7d;
        add_header Cache-Control "public";
    }

    location ~* \.(js|css|html)$ {
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header Pragma "no-cache";
        add_header Expires "0";
        try_files $uri $uri/ =404;
    }

    location / {
        try_files $uri $uri.html $uri/ /index.html;
    }
}

    }

    location /api/ {
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Tenant-Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_cache_bypass $http_upgrade;
    }

    location /socket.io/ {
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header X-Tenant-Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
    }

    location / {
        try_files $uri $uri.html $uri/ =404;
    }

    error_page 404 /404.html;
    location = /404.html {
        internal;
    }
}
