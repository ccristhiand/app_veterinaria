# ── API Backend ───────────────────────────────────────────────
server {
    listen 80;
    server_name api.netcodip.com;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name api.netcodip.com;

    ssl_certificate     /etc/letsencrypt/live/netcodip.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/netcodip.com/privkey.pem;

    location / {
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_cache_bypass $http_upgrade;
    }
}

# ── Places Scraper / Embudo Ventas CRM ───────────────────────
server {
    listen 80;
    server_name places-scraper.netcodip.com;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name places-scraper.netcodip.com;

    ssl_certificate     /etc/letsencrypt/live/netcodip.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/netcodip.com/privkey.pem;

    root /var/www/places-scraper/public;
    index index.html;

    # 1. Socket.io — PRIMERO, tiene prioridad sobre todo
    location ^~ /socket.io/ {
        proxy_pass http://localhost:3010;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_cache_bypass $http_upgrade;
        proxy_buffering off;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
        proxy_connect_timeout 60s;
    }

    # 2. API del backend
    location ^~ /api/ {
        proxy_pass http://localhost:3010;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 120s;
    }

    # 3. Imágenes subidas
    location ^~ /uploads/ {
        alias /var/www/places-scraper/uploads/;
        expires 7d;
        add_header Cache-Control "public";
    }

    # 4. Archivos estáticos (JS, CSS, HTML) — después de los proxies
    location ~* \.(js|css|html)$ {
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header Pragma "no-cache";
        add_header Expires "0";
        try_files $uri $uri/ =404;
    }

    # 5. Todo lo demás
    location / {
        try_files $uri $uri.html $uri/ /index.html;
    }
}
